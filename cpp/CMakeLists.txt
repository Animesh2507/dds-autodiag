cmake_minimum_required(VERSION 3.16)
project(autodiag_cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(autodiag_core src/core.cpp)
target_include_directories(autodiag_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(autodiag_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(autodiag_core_cli src/main.cpp)
target_link_libraries(autodiag_core_cli PRIVATE autodiag_core)
find_package(behaviortree_cpp QUIET)
if(behaviortree_cpp_FOUND)
  # Upstream BehaviorTree.CPP exports BT::behaviortree_cpp; the ROS 2 package exports behaviortree_cpp::behaviortree_cpp.
  if(TARGET BT::behaviortree_cpp)
    set(AUTODIAG_BT_TARGET BT::behaviortree_cpp)
  else()
    set(AUTODIAG_BT_TARGET behaviortree_cpp::behaviortree_cpp)
  endif()
  add_executable(autodiag_bt src/bt_main.cpp)
  target_link_libraries(autodiag_bt PRIVATE autodiag_core ${AUTODIAG_BT_TARGET})
  target_compile_definitions(autodiag_bt PRIVATE AUTODIAG_WITH_BT=1)
  message(STATUS "BehaviorTree.CPP found: autodiag_bt enabled")
else()
  message(STATUS "BehaviorTree.CPP unavailable: build portable C++ core, BT executable disabled")
endif()
enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE autodiag_core)
add_test(NAME core_tests COMMAND test_core)
